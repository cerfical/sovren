find_package(Doxygen COMPONENTS dot)

if(NOT Doxygen_FOUND)
    message(WARNING "Doxygen not found: documentation disabled")
    return()
endif()

# Doxygen configuration
set(DOXYGEN_REPEAT_BRIEF NO)
set(DOXYGEN_HIDE_UNDOC_MEMBERS YES)
set(DOXYGEN_HIDE_UNDOC_CLASSES YES)
set(DOXYGEN_WARN_IF_UNDOCUMENTED NO)

set(DOXYGEN_BUILTIN_STL_SUPPORT YES)
set(DOXYGEN_INLINE_SIMPLE_STRUCTS YES)
set(DOXYGEN_EXTRACT_PRIV_VIRTUAL YES)
set(DOXYGEN_INLINE_INHERITED_MEMB YES)

set(DOXYGEN_SHOW_HEADERFILE NO)
set(DOXYGEN_SHOW_INCLUDE_FILES NO)
set(DOXYGEN_INLINE_INFO NO)
set(DOXYGEN_COLLABORATION_GRAPH NO)

set(DOXYGEN_SORT_MEMBER_DOCS YES)
set(DOXYGEN_SORT_BRIEF_DOCS YES)
set(DOXYGEN_SORT_MEMBERS_CTORS_1ST YES)

set(DOXYGEN_SHOW_USED_FILES NO)
set(DOXYGEN_SHOW_FILES NO)

list(APPEND DOXYGEN_STRIP_FROM_INC_PATH "${CMAKE_SOURCE_DIR}/include")
set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "${CMAKE_SOURCE_DIR}/README.md")

# Generate documentation
get_target_property(SOVREN_LIB_HEADERS sovren-lib HEADER_SET)

doxygen_add_docs(sovren-docs ALL USE_STAMP_FILE
    "${SOVREN_LIB_HEADERS}"
    "${CMAKE_SOURCE_DIR}/README.md"
)
